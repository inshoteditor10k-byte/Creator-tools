Channel ownership & uniqueness checks (Prevents multiple accounts from claiming the same channel). * 3. Razorpay secure server-side Order creation. * 4. Razorpay HMAC-SHA256 signature verification. * 5. Campaign moderation management and performance metrics (real impression and click recording). */ const functions = require("firebase-functions"); const admin = require("firebase-admin"); const express = require("express"); } else { ytUrl = `https://www.googleapis.com/youtube/v3/channels?part=snippet,statistics&id=${channelId}&key=${YOUTUBE_API_KEY}`; } const ytRes = await axios.get(ytUrl); if (!ytRes.data.items || ytRes.data.items.length === 0) { return res.status(404).json({ message: "No matching YouTube Channel found." }); } const ch = ytRes.data.items[0]; const thumb = (ch.snippet.thumbnails.high || ch.snippet.thumbnails.medium || ch.snippet.thumbnails.default).url; // Check if channel is already claimed by another user in Firestore const existingClaim = await db.collection("channels").doc(ch.id).get(); if (existingClaim.exists && existingClaim.data().ownerUid !== req.user.uid) { return res.status(409).json({ message: "This YouTube Channel is already claimed by another PLAYVERSE user." }); } return res.status(200).json({ id: ch.id, title: ch.snippet.title, description: ch.snippet.description, customUrl: ch.snippet.customUrl || `@${ch.snippet.title.replace(/\s+/g, '').toLowerCase()}`, thumbnail: thumb, subscriberCount: ch.statistics.subscriberCount || 0, videoCount: ch.statistics.videoCount || 0, viewCount: ch.statistics.viewCount || 0, }); } catch (error) { console.error("YouTube Channel Verification Error:", error.response ? error.response.data : error.message); return res.status(500).json({ message: "Error communicating with YouTube Data API." }); } }); // ================= ENDPOINT 3: PERMANENTLY CLAIM CHANNEL ================= app.post("/claim-channel", authenticateFirebaseToken, async (req, res) => { try { const { channelData } = req.body; if (!channelData || !channelData.id) { return res.status(400).json({ message: "Invalid channel payload." }); } const options = { amount: Math.round(amountINR * 100), // In Paise currency: "INR", receipt: `pv_ad_${Date.now()}_${req.user.uid.slice(0, 5)}`, notes: { creatorUid: req.user.uid, creatorEmail: req.user.email, }, };
